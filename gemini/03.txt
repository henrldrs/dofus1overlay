  <script>
    // State management
    let currentItemId = null;
    let navigationStack = [];

    const searchInput = document.getElementById('search-input');
    const searchView = document.getElementById('search-view');
    const itemView = document.getElementById('item-view');
    const searchResultsList = document.getElementById('search-results-list');
    const craftQtyInput = document.getElementById('craft-qty');
    const btnBack = document.getElementById('btn-back');
    const agentInput = document.getElementById('agent-input');
    const btnAskAgent = document.getElementById('btn-ask-agent');

    // Controls
    document.getElementById('btn-close').addEventListener('click', () => window.close());
    document.getElementById('btn-minimize').addEventListener('click', () => {
      if (window.overlayAPI && window.overlayAPI.minimize) window.overlayAPI.minimize();
    });

    // Real-time FTS Search Listener
    searchInput.addEventListener('input', async (e) => {
      const query = e.target.value.trim();
      if (query.length < 2) {
        showSearchView();
        searchResultsList.innerHTML = '<div style="color: var(--text-muted); padding: 10px; text-align: center;">Type at least 2 characters...</div>';
        return;
      }

      showSearchView();
      try {
        const results = await window.overlayAPI.searchItems(query);
        renderSearchResults(results);
      } catch (err) {
        searchResultsList.innerHTML = `<div style="color: var(--stat-red); padding: 8px;">Backend connection error. Ensure Flask API is running.</div>`;
      }
    });

    function showSearchView() {
      searchView.classList.remove('hidden');
      itemView.classList.add('hidden');
    }

    function renderSearchResults(items) {
      if (!items || items.length === 0) {
        searchResultsList.innerHTML = '<div style="color: var(--text-muted); padding: 10px; text-align: center;">No items found.</div>';
        return;
      }

      searchResultsList.innerHTML = items.map(item => `
        <div class="result-card" onclick="navigateToItem(${item.id})">
          <div class="result-info">
            <span class="result-name">${escapeHtml(item.name)}</span>
            <span class="result-meta">${escapeHtml(item.type)}</span>
          </div>
          <span class="level-badge">Lvl ${item.level}</span>
        </div>
      `).join('');
    }

    // Item Detail Navigator
    async function navigateToItem(id, pushHistory = true) {
      if (pushHistory && currentItemId) {
        navigationStack.push(currentItemId);
      }
      currentItemId = id;

      searchView.classList.add('hidden');
      itemView.classList.remove('hidden');
      document.getElementById('history-indicator').innerText = navigationStack.length > 0 ? `(${navigationStack.length} back)` : '';

      try {
        const item = await window.overlayAPI.getItemDetail(id);
        renderItemDetail(item);
        loadRecipeBreakdown(id, parseInt(craftQtyInput.value) || 1);
      } catch (err) {
        console.error("Error loading item details:", err);
      }
    }

    function renderItemDetail(item) {
      document.getElementById('detail-name').innerText = item.name;
      document.getElementById('detail-type').innerText = item.type;
      document.getElementById('detail-level').innerText = item.level;
      document.getElementById('detail-pods').innerText = item.pods || 0;
      
      const iconImg = document.getElementById('detail-icon');
      iconImg.src = item.icon_url || 'https://solomonk.fr/assets/items/0.png';
      iconImg.onerror = () => { iconImg.src = 'data:image/svg+xml;utf8,<svg xmlns="http://www.w3.org/2000/svg" width="44" height="44" viewBox="0 0 24 24" fill="none" stroke="%238c93a8" stroke-width="2"><rect x="3" y="3" width="18" height="18" rx="2"/><circle cx="8.5" cy="8.5" r="1.5"/><path d="M21 15l-5-5L5 21"/></svg>'; };

      // Description
      const descBlock = document.getElementById('desc-block');
      if (item.description) {
        descBlock.classList.remove('hidden');
        document.getElementById('detail-desc').innerText = item.description;
      } else {
        descBlock.classList.add('hidden');
      }

      // Stats
      const statsBlock = document.getElementById('stats-block');
      const statsContainer = document.getElementById('detail-stats');
      if (item.stats && item.stats.length > 0) {
        statsBlock.classList.remove('hidden');
        statsContainer.innerHTML = item.stats.map(s => `
          <div class="stat-item">
            <span>${escapeHtml(s.stat || s.name || '')}</span>
            <span class="stat-val">${escapeHtml(String(s.value || s.max || ''))}</span>
          </div>
        `).join('');
      } else {
        statsBlock.classList.add('hidden');
      }

      // Drops
      const dropBlock = document.getElementById('drop-block');
      const dropContainer = document.getElementById('detail-drops');
      if (item.drops && item.drops.length > 0) {
        dropBlock.classList.remove('hidden');
        dropContainer.innerHTML = item.drops.map(d => `
          <div class="drop-row">
            <span>${escapeHtml(d.monster_name)}</span>
            <span class="drop-rate">${d.drop_rate}%</span>
          </div>
        `).join('');
      } else {
        dropBlock.classList.add('hidden');
      }
    }

    // Dynamic Multiplier Recalculator
    async function loadRecipeBreakdown(id, qty) {
      const recipeBlock = document.getElementById('recipe-block');
      const recipeContainer = document.getElementById('detail-recipe');

      try {
        const breakdown = await window.overlayAPI.getRecipeBreakdown(id, qty);
        if (!breakdown || !breakdown.raw_materials || breakdown.raw_materials.length === 0) {
          recipeBlock.classList.add('hidden');
          return;
        }

        recipeBlock.classList.remove('hidden');
        recipeContainer.innerHTML = breakdown.raw_materials.map(mat => `
          <div class="ingredient-row" ${mat.id ? `onclick="navigateToItem(${mat.id})"` : ''}>
            <span>${escapeHtml(mat.name)}</span>
            <span class="ing-qty">x${mat.quantity}</span>
          </div>
        `).join('');
      } catch (e) {
        recipeBlock.classList.add('hidden');
      }
    }

    // Quantity Change Listener
    craftQtyInput.addEventListener('input', (e) => {
      const qty = Math.max(1, parseInt(e.target.value) || 1);
      if (currentItemId) {
        loadRecipeBreakdown(currentItemId, qty);
      }
    });

    // History Back Navigation
    btnBack.addEventListener('click', () => {
      if (navigationStack.length > 0) {
        const previousId = navigationStack.pop();
        navigateToItem(previousId, false);
      } else {
        showSearchView();
      }
    });

    // Local Qwen Agent Querying
    btnAskAgent.addEventListener('click', sendAgentQuery);
    agentInput.addEventListener('keypress', (e) => {
      if (e.key === 'Enter') sendAgentQuery();
    });

    async function sendAgentQuery() {
      const prompt = agentInput.value.trim();
      if (!prompt) return;

      btnAskAgent.innerText = "...";
      try {
        const response = await fetch('http://127.0.0.1:5000/api/search?q=' + encodeURIComponent(prompt));
        const resJson = await response.json();
        if (resJson.length > 0) {
          navigateToItem(resJson[0].id);
        }
      } catch (err) {
        console.error("Agent query error:", err);
      } finally {
        btnAskAgent.innerText = "Ask";
      }
    }

    function escapeHtml(str) {
      return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
    }
  </script>
